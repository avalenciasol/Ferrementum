create database visa_project
create table aspirante
(
Documento varchar (45) not null PRIMARY KEY, 
Tipo_doc varchar (50) not null, 
Nombre varchar (50) not null, 
Apellido varchar (45) not null, 
Email varchar (60) not null, 
Nacionalidad varchar (45) not null, 
Edad int not null,
Pasaporte varchar (45) not null,
Estado_soli varchar (45) not null, 
Id_visa int not null 
 foreign key (Id_visa) references visa (Id_visa)
)
CREATE TABLE visa
(
Id_visa INT NOT NULL PRIMARY KEY, 
Tipo_visa varchar (45) not null,
fecha_exp date not null, 
fecha_ven date not null, 
estado varchar (30) not null
)
create TABLE requisitos (
Id_requisitos int not null primary key,
Estado_requisitos varchar (20) not null, 
observacion Varchar (60) not null, 
Id_visa int
foreign key (Id_visa) references visa (Id_visa)
)

CREATE TABLE Entrevista  (
Id_entrevista int not null primary key, 
Orden_pago int not null, 
Id_canciller Varchar (45) not null, 
Entrevista date not null, 
Estado_requisitos int
foreign key (Estado_requisitos) references requisitos (Id_requisitos)
)

SELECT a.Tipo_doc, r.observacion, a.Nombre, a.Apellido, v.Tipo_visa
FROM aspirante a
INNER JOIN requisitos r ON a.Id_visa = r.Id_visa
INNER JOIN visa v ON a.Id_visa = v.Id_visa;

SELECT a.Documento, a.Nombre, a.Apellido, v.Tipo_visa
FROM aspirante a

---------------------------
SELECT a.Documento, a.Nombre, a.Apellido, v.Tipo_visa
FROM aspirante a
LEFT JOIN visa v ON a.Id_visa = v.Id_visa;
--------------------------
SELECT a.Documento, a.Nombre, a.Apellido, v.Tipo_visa
FROM aspirante a
RIGHT JOIN visa v ON a.Id_visa = v.Id_visa;
------------------------
SELECT v.Tipo_visa, COUNT(*) AS Cantidad_Aspirantes
FROM aspirante a
LEFT JOIN visa v ON a.Id_visa = v.Id_visa
GROUP BY v.Tipo_visa;
